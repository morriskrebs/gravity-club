let metaPixelLoaded = false;


function initMetaPixel() {

  !(function(f, b, e, v, n, t, s) {

    if (f.fbq) return;

      n.callMethod ? n.callMethod.apply(n, arguments) : n.queue.push(arguments);

    };

    if (!f._fbq) f._fbq = n;

    n.push = n;

    n.loaded = true;

    n.version = '2.0';

    n.queue = [];

    t = b.createElement(e);

async function play() {

  const firstName = String(document.getElementById('firstName')?.value || '').trim();

  const lastName = String(document.getElementById('lastName')?.value || '').trim();

  const marketingConsent = Boolean(document.getElementById('marketingConsent')?.checked);


  showMessage('');


  if (!firstName) {

    trackGtagEvent('error_firstname');

    showMessage('ENTER YOUR FIRST NAME');

    return;

  }


  if (!lastName) {

    trackGtagEvent('error_lastname');

    showMessage('ENTER YOUR LAST NAME');

    return;

  }


  if (!email || !email.includes('@')) {

    trackGtagEvent('error_email');

    showMessage('ENTER A VALID EMAIL');

    return;

  }


  if (!marketingConsent) {

    trackGtagEvent('error_consent');

    showMessage('PLEASE ACCEPT TO CONTINUE');

    return;

  }


  trackGtagEvent('submit_valid');


  const existingWinner = findWinnerByEmail(email);

  if (existingWinner) {

    showMessage(buildWinnerMessage(), true);

    trackGtagEvent('win');

    return;

  }


  setLoader(true);

  await new Promise(resolve => setTimeout(resolve, 1200));


  let win = Math.random() < 0.3;

  if (email === 'morris.krebs@gmail.com') {

    win = true;

  }


  const submissionPayload = {

    firstName,

    lastName,

    email,

    marketingConsent: marketingConsent ? 'yes' : 'no',

    status: win ? 'winner' : 'loser',

    source: 'street_qr',

    submittedAt: new Date().toISOString()

  };


  try {

    await submitWinnerToFormspree(submissionPayload);


    if (!win) {

      const losses = Number(localStorage.getItem('gc_losses') || 0) + 1;

      localStorage.setItem('gc_losses', String(losses));

      setLoader(false);

      showMessage(buildLoseMessage());

      trackGtagEvent('lose');

      return;

    }


    const winnerCode = 'GC-' + Math.floor(1000 + Math.random() * 9000);

    const wins = Number(localStorage.getItem('gc_wins') || 0) + 1;

    localStorage.setItem('gc_wins', String(wins));

    saveWinner(email, winnerCode);

    trackMetaEvent('Lead', { content_name: 'Gravity Club Free Class' });

    trackGtagEvent('generate_lead');

    trackGtagEvent('win');

    setLoader(false);

    showMessage(buildWinnerMessage(), true);


  } catch (error) {

    setLoader(false);

    showMessage('TRY AGAIN');

  }

}


function openLegalOverlay(type) {

  const overlay = document.getElementById('legalOverlay');

  const title = document.getElementById('legalTitle');

  const body = document.getElementById('legalBody');

  const entry = LEGAL_CONTENT[type];

  if (!overlay || !title || !body || !entry) return;


  title.textContent = entry.title;

  body.innerHTML = entry.body.replace(/\n/g, '<br>');

  overlay.classList.remove('consent-hidden');

}


function closeLegalOverlay() {

  const overlay = document.getElementById('legalOverlay');

  if (overlay) {

    overlay.classList.add('consent-hidden');

  }

}
